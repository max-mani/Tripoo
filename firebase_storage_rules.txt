rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // Helper function to check if user is authenticated
    function isAuthenticated() {
      return request.auth != null;
    }
    
    // Profile images - path structure: profile_images/{userId}_{uuid}.jpg
    // Allow authenticated users to read any profile image
    // Users can only upload/update/delete their own images
    match /profile_images/{fileName} {
      // Extract userId from filename (format: userId_uuid.jpg)
      function getUserIdFromFileName() {
        return fileName.split('_')[0];
      }
      
      // Allow read for all authenticated users
      allow read: if isAuthenticated();
      
      // Allow write only if the userId in filename matches authenticated user
      allow write: if isAuthenticated() && request.auth.uid == getUserIdFromFileName();
    }
    
    // Allow authenticated users to read/write other files if needed
    // Remove or restrict this if you don't need other storage paths
    match /{allPaths=**} {
      allow read, write: if isAuthenticated();
    }
  }
}
